#include <Servo.h>

const int TRIG_PIN = 9;
const int ECHO_PIN = 10;
const int SERVO_PIN = 11;

Servo radarServo;

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  radarServo.attach(SERVO_PIN);
}

void loop() {
  for (int pos = 0; pos <= 180; pos += 2) {
    radarServo.write(pos);
    delay(15);
    long distance = readDistance();
    Serial.print(pos);
    Serial.print(",");
    Serial.println(distance);
  }
}

Common Questions

Frequently Asked Questions

Everything you need to know about setting up and programming with Arduino Claw.

Can I use Arduino Claw completely offline?

Yes! By pointing the AI Settings to a local Ollama or LM Studio instance running models like qwen2.5-coder or codellama, you can prompt, generate code, compile, and flash entirely offline without sending any code over the internet.

Do I need to install the official Arduino IDE first?

No. Arduino Claw bundles its own internal compiler runtime, toolchains, and USB communication bridges. You do not need the Arduino IDE installed on your computer.

How does in-app USB flashing work?

Arduino Claw communicates over your computer's USB serial controller using native STK500v1 serial flashing routines. When you click Flash, it opens the selected serial port, triggers the microcontroller DTR auto-reset, and streams the compiled hex payload directly into flash memory.

Which AI providers and models can I use?

Arduino Claw supports Google Gemini (Gemini 2.0 Flash / 1.5 Pro), Groq (Llama 3.3 70B), Cerebras, SambaNova, OpenRouter (DeepSeek R1, Claude 3.5 Sonnet, GPT-4o), as well as local Ollama and LM Studio endpoints.

Is Arduino Claw free and open source?

Yes. Arduino Claw is 100% open source and licensed under the permissive MIT License. You can inspect the source code, fork the project, and contribute on GitHub.
